  (defun my/set-mermaid-cli-path ()
  "Sets `org-mermaid-cli-path` based on the operating system.
   Uses `which` on Linux/WSL and `where` on Windows."
  (cond
   ;; Case 1: Linux (including WSL Ubuntu)
   ((or (eq system-type 'gnu/linux) (eq system-type 'cygwin))
    (let ((mmdc-path (shell-command-to-string "which mmdc")))
      (setq mmdc-path (string-trim mmdc-path)) ; Remove newline/whitespace
      (if (and (not (string-empty-p mmdc-path)) (file-exists-p mmdc-path))
	  (progn
	    (setq org-mermaid-cli-path mmdc-path)
	    (message "Mermaid CLI path (Linux/WSL) set to: %s" org-mermaid-cli-path))
	(message "WARNING: 'mmdc' not found in PATH on Linux/WSL. Please ensure it's installed via Nix/npm and in your system's PATH."))))

   ;; Case 2: Windows
   ((eq system-type 'windows-nt)
    (let ((mmdc-path (shell-command-to-string "where mmdc.cmd"))) ; Use .cmd for Windows executables
      (setq mmdc-path (string-trim mmdc-path))
      (if (and (not (string-empty-p mmdc-path)) (file-exists-p mmdc-path))
	  (progn
	    (setq org-mermaid-cli-path mmdc-path)
	    (message "Mermaid CLI path (Windows) set to: %s" org-mermaid-cli-path))
	(message "WARNING: 'mmdc.cmd' not found in PATH on Windows. Please install 'mermaid-cli' via npm and ensure Node.js/npm is in your PATH."))))

   ;; Case 3: Other or Unknown OS
   (t
    (message "Mermaid CLI: Unsupported system-type '%s' for automatic path detection." system-type))))

  (cond
   ((eq system-type 'gnu/linux)
    (my/set-mermaid-cli-path)
    (setq ob-mermaid-cli-path "/mnt/c/Users/shunt/AppData/Roaming/npm/mmdc")))
